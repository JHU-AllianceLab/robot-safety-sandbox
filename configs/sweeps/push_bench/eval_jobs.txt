# The SONIC push benchmark rebuilt in mjlab -- the evaluations (README.md). Per seed, in order:
#   SHAC: 1. calibration records on the calibration table (2048 trials, seed 1), 2. the conformal
#         worst-margin tube, 3. his dev-heldout table (data/zz_test/trials.csv, sha256 07cdc6bf...),
#         4. the training-motion table; conditions: SONIC, SHAC alone, the WM tube (1 s, c 0.95)
#         plain / dwell 2 s / latch.
#   SafePPO: 5./6. the same two tables; conditions: SONIC, SafePPO alone, the exact-sim filter (1 s).
# Array task N runs the jobs of seed N-1 (eval_seed.sh), one GPU and 32 CPUs (the rollout pool):
#   sbatch -A hhu49_alliance1 -p rtx6000,l40s,h100 --array=1-3 --gres=gpu:1 --cpus-per-task=32 --mem=96G \
#       --time=24:00:00 --job-name=push_bench_eval --output=logs/slurm/%x_%A_%a.out configs/sweeps/push_bench/eval_seed.sh
python examples/eval_push_bench.py --table data/push_bench/calib_v1/trials.csv --sample 2048 --no-controls --seed 1 --policy runs/push_bench/shac_s0/g1_recover_bench_safetyshac1p/final_model.zip --tag shac_s0 --sonic-dir $HOME/gear_sonic_hf --conditions wmcalib:clean --out runs/push_bench/calib/shac_s0  # creates: runs/push_bench/calib/shac_s0/bench.pt
python examples/calibrate_wm_min_tube.py --calib runs/push_bench/calib/shac_s0/bench.pt --tag shac_s0  # creates: runs/push_bench/calib/shac_s0/min_tube.json
python examples/eval_push_bench.py --table data/zz_test/trials.csv --policy runs/push_bench/shac_s0/g1_recover_bench_safetyshac1p/final_model.zip --tag shac_s0 --sonic-dir $HOME/gear_sonic_hf --conditions sonic fallback wmmin:1:0.95 wmmin:1:0.95+dwell2 wmmin:1:0.95+latch --tube-calib runs/push_bench/calib/shac_s0/min_tube.json --out runs/push_bench/eval/dev/shac_s0
python examples/eval_push_bench.py --table data/push_bench/train_v1/trials.csv --policy runs/push_bench/shac_s0/g1_recover_bench_safetyshac1p/final_model.zip --tag shac_s0 --sonic-dir $HOME/gear_sonic_hf --conditions sonic fallback wmmin:1:0.95 wmmin:1:0.95+dwell2 wmmin:1:0.95+latch --tube-calib runs/push_bench/calib/shac_s0/min_tube.json --out runs/push_bench/eval/train/shac_s0
